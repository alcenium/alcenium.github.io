  void display();
  void buzzerOn();
  void handleBack();
  void handleEnter();
  void openTrashcan();
  
  // ========== Khởi tạo struct ====================
  typedef struct menuItem MenuItem;

  typedef struct {
    MenuItem *itemList;
    int length;
  } Menu;

  struct menuItem
  {
    const char *label;
    void (*action)();
    Menu *submenu;
  };

  // ========== Tạo thành phần trong menu ==========

  Menu dsThanhVien = {dsThanhVienItems, 5};

  MenuItem features[] =
  {{"Mo nap", openTrashcan, NULL},
   {"Bat loa", buzzerOn, NULL}};
  Menu featuresMenu = {features, 2};

  MenuItem mainMenuItems[] =
  {{"Thanh vien", NULL, &dsThanhVien},
   "Tinh nang", NULL, &featuresMenu};
  Menu mainMenu = {mainMenuItems, 2};
  
  
  // ========== Tạo con trỏ đến các module ngoài ==========
  LiquidCrystal_I2C lcd(0x27, 20, 4);
  Servo myServo;
  
  
  // ========== Đặt tên cho các cổng ======================
  const int bUp = 2, bDown = 3, bEnter = 4, bBack = 5;
  const int ledFull = 6, ledEmpty = 7, ledFire = 13;
  
  const int ultraSonicEcho = 8, ultraSonicTrig = 9;
  const int servo = 10;
  
  const int fireSensor = 11, fullSensor = 12;
  
  
  // ========== Stack cho menu ============================
  Menu *menuStack[3];
  int topStack = 0;
  Menu *currentMenu;

  int selectedIndex = 0;
  int scrollOffset = 0;
  
  
  // ========== Cảm biến khoảng cách =======================
  unsigned long duration;
  int distance;
  
  
  // ========== Timer ======================================
  unsigned long ultraSonicLastExecuted = 0,
                servoLastExecuted = 0;
  
  bool servoOpened = false;
  
  void setup() {
    menuStack[topStack] = &mainMenu;
    currentMenu = menuStack[topStack];
  
    Serial.begin(9600);
    lcd.init();
    lcd.backlight();
  
    myServo.attach(servo);
    myServo.write(0);

    pinMode(ultraSonicEcho, INPUT);
    pinMode(ultraSonicTrig, OUTPUT);
  
    pinMode(fullSensor, INPUT);
    pinMode(fireSensor, INPUT);
  
    pinMode(bUp, INPUT_PULLUP);
    pinMode(bDown, INPUT_PULLUP);
    pinMode(bEnter, INPUT_PULLUP);
    pinMode(bBack, INPUT_PULLUP);
  
    pinMode(ledFull, OUTPUT);
    pinMode(ledEmpty, OUTPUT);
    pinMode(ledFire, OUTPUT);
    display();
  }
  
  
  void loop() {
    // ========== Cảm biến ===================
    if (digitalRead(fullSensor) == 0) {
      digitalWrite(ledFull, HIGH);
      digitalWrite(ledEmpty, LOW);
    } else {
      digitalWrite(ledFull, LOW);
      digitalWrite(ledEmpty, HIGH);
    }
  
    if (digitalRead(fireSensor) == 0) {
      digitalWrite(ledFire, HIGH);
    } else {
      digitalWrite(ledFire, LOW);
    }
  
  
    // ========== Tự động mở nắp =============
    if (millis() - ultraSonicLastExecuted >= 500) {
      digitalWrite(ultraSonicTrig, 0);
      delayMicroseconds(2);
      digitalWrite(ultraSonicTrig, 1);
      delayMicroseconds(5);
      digitalWrite(ultraSonicTrig, 0);

      duration = pulseIn(ultraSonicEcho, HIGH, 30000);
      distance = int (duration / 2 / 29.412);

      Serial.print(distance);
      Serial.println(" cm");

      if (distance > 0 && distance <= 15)
        openTrashcan();

      ultraSonicLastExecuted = millis();
    }

    if (servoOpened == true && millis() - servoLastExecuted >= 3000){
      myServo.write(0);
      servoOpened = false;
    }
  
  
    // ========== Bluetooth ==================
    if (Serial.available() > 0) {
      char c = Serial.read();

      switch (c) {
        case 'b':
          buzzerOn();
          break;
        case 't':
          openTrashcan();
          break;
        case '\n':
        case '\r':
          break;
        default:
          Serial.print("Ko hiểu lệnh: ");
          Serial.print(c);
          break;
      }
    }
  
  
    // ========== Chức năng của nút ==========
    if (!digitalRead(bUp)) {
      if (selectedIndex > 0) selectedIndex--;
      if (selectedIndex < scrollOffset) scrollOffset--;
      Serial.println(selectedIndex);
      Serial.println(scrollOffset);
      display();

      delay(100);
      while (!digitalRead(bUp));
    }

    if (!digitalRead(bDown)) {
      if (selectedIndex < currentMenu->length-1) selectedIndex++;
      if (selectedIndex > scrollOffset+2) scrollOffset++;
      Serial.println(selectedIndex);
      Serial.println(scrollOffset);
      display();

      delay(100);
      while (!digitalRead(bDown));
    }

    if (!digitalRead(bEnter)) {
      handleEnter();

      delay(100);
      while (!digitalRead(bEnter));
    }

    if (!digitalRead(bBack)) {
      handleBack();
      display();

      delay(100);
      while(!digitalRead(bBack));
    }
  }
  
  
  // ========== Các chức năng cho module ==========
  void display() {
    lcd.clear();
    int menuLength = currentMenu->length;
    lcd.setCursor(0, 0);
    lcd.print("NHOM 8 -LOP 74DCTT21");

    Serial.println(menuLength);

    // Menu
    for (int i=scrollOffset, line=1; line<=3 && line<menuLength+1; line++, i++) {
      lcd.setCursor(0, line);
      lcd.print(currentMenu->itemList[i].label);
      Serial.println(currentMenu->itemList[i].label);
    }

    // Selection
    lcd.setCursor(19, selectedIndex-scrollOffset+1);
    lcd.print('<');
  }
  
  
  void handleEnter() {
    MenuItem chosenItem = currentMenu->itemList[selectedIndex];
  
    if (chosenItem.action != NULL)
      chosenItem.action();
  
    if (chosenItem.submenu != NULL) {
      currentMenu = chosenItem.submenu;
      menuStack[++topStack] = currentMenu;
      selectedIndex = 0;
      scrollOffset = 0;
      display();
    }
  }
  
  
  void handleBack() {
    if (topStack == 0) return;
  
    menuStack[topStack--] = NULL;
    currentMenu = menuStack[topStack];
    selectedIndex = 0;
    scrollOffset=0;
  }


  void buzzerOn() {
    digitalWrite(ledFire, HIGH);
    delay(1000);
    digitalWrite(ledFire, LOW);
  }

  void openTrashcan() {
    Serial.println("Servo running");
    myServo.write(90);
    servoOpened = true;
    servoLastExecuted = millis();
  }
}
